🔗 Connect Google Sheets

Your data will be stored in a Google Sheet — accessible from any device. Follow these steps to set it up (one-time, takes ~3 minutes):

  1. Open Google Sheets and create a new spreadsheet.
  2. Go to Extensions → Apps Script.
  3. Delete the default code and paste the script below.
  4. Click Deploy → New deployment → Web app.
  5. Set "Who has access" to Anyone, then click Deploy.
  6. Copy the Web app URL and paste it below.

Apps Script to paste:

// Daily Routine Dashboard — Apps Script Backend const SHEET_NAME_TASKS = 'Tasks'; const SHEET_NAME_REC = 'Recurring'; function getOrCreateSheet(ss, name, headers) { let sh = ss.getSheetByName(name); if (!sh) { sh = ss.insertSheet(name); sh.appendRow(headers); sh.setFrozenRows(1); } return sh; } function doGet(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const tSh = getOrCreateSheet(ss, SHEET_NAME_TASKS, ['id','name','date','type','done','startTime','endTime','category','notes','recurringId','recurringDaysLeft']); const rSh = getOrCreateSheet(ss, SHEET_NAME_REC, ['id','name','startDate','totalDays','daysLeft','duration','startTime','endTime','category']); const tasks = sheetToArr(tSh); const rec = sheetToArr(rSh); return ContentService .createTextOutput(JSON.stringify({ tasks, recurring: rec })) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const body = JSON.parse(e.postData.contents); const { action, data } = body; const tSh = getOrCreateSheet(ss, SHEET_NAME_TASKS, ['id','name','date','type','done','startTime','endTime','category','notes','recurringId','recurringDaysLeft']); const rSh = getOrCreateSheet(ss, SHEET_NAME_REC, ['id','name','startDate','totalDays','daysLeft','duration','startTime','endTime','category']); if (action === 'saveAll') { clearAndWrite(tSh, ['id','name','date','type','done','startTime','endTime','category','notes','recurringId','recurringDaysLeft'], data.tasks, t => [t.id,t.name,t.date,t.type,t.done?1:0,t.startTime||'',t.endTime||'',t.category||'',t.notes||'',t.recurringId||'',t.recurringDaysLeft||'']); clearAndWrite(rSh, ['id','name','startDate','totalDays','daysLeft','duration','startTime','endTime','category'], data.recurring, r => [r.id,r.name,r.startDate,r.totalDays,r.daysLeft,r.duration||'',r.startTime||'',r.endTime||'',r.category||'']); } return ContentService .createTextOutput(JSON.stringify({ ok: true })) .setMimeType(ContentService.MimeType.JSON); } function sheetToArr(sh) { const rows = sh.getDataRange().getValues(); if (rows.length < 2) return []; const headers = rows[0]; return rows.slice(1).map(row => { const obj = {}; headers.forEach((h, i) => { obj[h] = row[i]; }); // coerce if ('done' in obj) obj.done = obj.done == 1 || obj.done === true || obj.done === 'TRUE'; if ('totalDays' in obj) obj.totalDays = Number(obj.totalDays) || 0; if ('daysLeft' in obj) obj.daysLeft = Number(obj.daysLeft) || 0; return obj; }); } function clearAndWrite(sh, headers, arr, rowFn) { const lastRow = sh.getLastRow(); if (lastRow > 1) sh.deleteRows(2, lastRow - 1); if (arr && arr.length) sh.getRange(2, 1, arr.length, headers.length).setValues(arr.map(rowFn)); }
